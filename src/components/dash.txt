import { useState, useEffect } from "react";
import { useNavigate, useLocation } from "react-router-dom";
import { Home, Package, Gift, User, PlusCircle } from "lucide-react";
import { useAuth } from "@/context/AuthContext";
import RequestForm from "@/components/RequestForm";
import HomeTab from "@/components/dashboard/HomeTab";
import BookingsTab from "@/components/dashboard/BookingsTab";
import ReferralsTab from "@/components/dashboard/ReferralsTab";
import ProfileTab from "@/components/dashboard/ProfileTab";

const TABS = [
  { id: "home", label: "Home", icon: Home },
  { id: "bookings", label: "Bookings", icon: Package },
  { id: "book", label: "Book", icon: PlusCircle },
  { id: "referrals", label: "Referrals", icon: Gift },
  { id: "profile", label: "Profile", icon: User },
];

export default function Dashboard() {
  const [activeTab, setActiveTab] = useState("home");
  const { user, token } = useAuth();
  const navigate = useNavigate();
  const location = useLocation();

  useEffect(() => {
    if (!token && !user && location.pathname === "/dashboard") {
      navigate("/auth");
    }
  }, [user, token, navigate, location.pathname]);

  if (!user && !token && location.pathname === "/dashboard") return null;

  const renderTab = () => {
    switch (activeTab) {
      case "home":
        return <HomeTab onTabChange={setActiveTab} />;
      case "bookings":
        return <BookingsTab />;
      case "book":
        return <RequestForm onTabChange={setActiveTab} />;
      case "referrals":
        return <ReferralsTab />;
      case "profile":
        return <ProfileTab />;
      default:
        return <HomeTab onTabChange={setActiveTab} />;
    }
  };

  return (
    <div className="min-h-screen bg-gray-50 font-poppins flex flex-col">
      {/* Top bar */}
      <header className="bg-white border-b border-gray-100 px-4 py-3 flex items-center justify-between sticky top-0 z-40">
        <div className="flex items-center gap-2">
          {/* <div className="w-7 h-7 rounded-full bg-[#127733] flex items-center justify-center">
            <span className="text-white text-xs font-bold">L</span>
          </div>
          <span className="font-semibold text-[#127733] text-sm">LaundryAid</span> */}
          <span className="font-semibold text-[#127733] text-sm">Dashboard</span> 
        
        </div>
        <span className="text-xs text-gray-400 capitalize">{activeTab}</span>
      </header>

      {/* Content */}
      <main className="flex-1 overflow-y-auto px-4 pt-4 pb-24 max-w-lg mx-auto w-full">
        {renderTab()}
      </main>

      {/* Bottom Navigation */}
      <nav className="fixed bottom-0 left-0 right-0 bg-white border-t border-gray-100 z-50 safe-area-pb">
        <div className="flex items-center justify-around max-w-lg mx-auto">
          {TABS.map(({ id, label, icon: Icon }) => {
            const isActive = activeTab === id;
            const isBook = id === "book";

            return (
              <button
                key={id}
                onClick={() => setActiveTab(id)}
                className={`flex flex-col items-center py-2 px-3 flex-1 transition-colors ${
                  isBook ? "" : ""
                }`}
              >
                {isBook ? (
                  <div
                    className={`w-12 h-12 rounded-full flex items-center justify-center -mt-5 shadow-lg transition-colors ${
                      isActive ? "bg-[#c85f0b]" : "bg-[#127733]"
                    }`}
                  >
                    <Icon size={22} className="text-white" />
                  </div>
                ) : (
                  <Icon
                    size={20}
                    className={isActive ? "text-[#127733]" : "text-gray-400"}
                  />
                )}
                <span
                  className={`text-[10px] mt-0.5 ${
                    isBook ? "mt-1" : ""
                  } ${isActive && !isBook ? "text-[#127733] font-medium" : "text-gray-400"}`}
                >
                  {label}
                </span>
              </button>
            );
          })}
        </div>
      </nav>
    </div>
  );
}
